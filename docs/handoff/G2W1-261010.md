```sql
SELECT status,count(*),count(output) AS outputs
FROM public.dashboard_generation_runs GROUP BY status;
SELECT count(*) AS retained_without_current_consent
FROM public.dashboard_generation_runs g
WHERE g.output IS NOT NULL AND NOT EXISTS (
  SELECT 1 FROM public.users u
  CROSS JOIN LATERAL public.llm_consent_current_decision(u.id) c
  WHERE u.id=g.user_id AND u.privacy_prefs->'recommendations'='true'::jsonb
    AND c.allowed AND c.token=g.consent_token
    AND NOT EXISTS (SELECT 1 FROM public.consent_changes e WHERE e.user_id=u.id
      AND e.pref_key='recommendations' AND e.event_type='revoke' AND e.created_at>=g.created_at)
);
SELECT purpose,reasoning_effort,model_used,count(*)
FROM public.ai_audit_log WHERE outbox_event_id LIKE 'dashboard:%'
GROUP BY purpose,reasoning_effort,model_used;
SELECT tgname,tgenabled FROM pg_trigger
WHERE tgname IN ('zz_dashboard_privacy_changed','dashboard_receipt_changed',
  'dashboard_consent_record_changed','zz_dashboard_consent_event_changed','dashboard_consent_event_lock');
SELECT p.proname,p.prosecdef,p.proconfig,
  has_function_privilege('anon',p.oid,'EXECUTE') AS anon_execute,
  has_function_privilege('authenticated',p.oid,'EXECUTE') AS client_execute,
  has_function_privilege('service_role',p.oid,'EXECUTE') AS server_execute
FROM pg_proc p JOIN pg_namespace n ON n.oid=p.pronamespace
WHERE n.nspname='public' AND p.proname IN ('dashboard_generation_audit_attempt',
  'dashboard_generation_audit_result','invalidate_dashboard_consent','lock_dashboard_consent_event');
```

새 audit RPC 둘은 server_execute만 true, 내부 trigger 함수 둘은 세 역할 모두 false여야 한다.
